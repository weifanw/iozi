cmake_minimum_required(VERSION 3.20)
project(iozi VERSION 0.1.0 LANGUAGES CXX)

option(IOZI_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(iozi_core INTERFACE)
target_include_directories(iozi_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(iozi_core INTERFACE cxx_std_20)

add_library(iozi_options INTERFACE)
if(MSVC)
    target_compile_options(iozi_options INTERFACE /W4 /permissive- /utf-8)
else()
    target_compile_options(iozi_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(IOZI_SANITIZE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
        target_compile_options(iozi_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(iozi_options INTERFACE -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "IOZI_SANITIZE requires GCC or Clang with sanitizer support")
    endif()
endif()

add_library(iozi_corpus src/corpus.cpp)
target_link_libraries(iozi_corpus PUBLIC iozi_core PRIVATE iozi_options)
add_executable(iozi src/main.cpp)
target_link_libraries(iozi PRIVATE iozi_corpus iozi_options)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_engine tests/test_engine.cpp)
    target_link_libraries(test_engine PRIVATE iozi_core iozi_options)
    add_executable(test_corpus tests/test_corpus.cpp)
    target_link_libraries(test_corpus PRIVATE iozi_corpus iozi_options)
    add_test(NAME engine COMMAND test_engine)
    add_test(NAME corpus COMMAND test_corpus)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:iozi>)
endif()
