cmake_minimum_required(VERSION 3.20)
project(minse VERSION 0.1.0 LANGUAGES CXX)

option(MINSE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(minse_core INTERFACE)
target_include_directories(minse_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(minse_core INTERFACE cxx_std_20)

add_library(minse_options INTERFACE)
if(MSVC)
    target_compile_options(minse_options INTERFACE /W4 /permissive- /utf-8)
else()
    target_compile_options(minse_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(MINSE_SANITIZE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
        target_compile_options(minse_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(minse_options INTERFACE -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "MINSE_SANITIZE requires GCC or Clang with sanitizer support")
    endif()
endif()

add_library(minse_corpus src/corpus.cpp)
target_link_libraries(minse_corpus PUBLIC minse_core PRIVATE minse_options)
add_executable(minse src/main.cpp)
target_link_libraries(minse PRIVATE minse_corpus minse_options)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_engine tests/test_engine.cpp)
    target_link_libraries(test_engine PRIVATE minse_core minse_options)
    add_executable(test_corpus tests/test_corpus.cpp)
    target_link_libraries(test_corpus PRIVATE minse_corpus minse_options)
    add_test(NAME engine COMMAND test_engine)
    add_test(NAME corpus COMMAND test_corpus)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:minse>)
endif()
